% Self-contained Latin and CJK typography.
% Fonts are distributed in assets/fonts/ with their original licenses.

\defaultfontfeatures{Ligatures=TeX,Scale=MatchLowercase}

\setmainfont[
  Path=assets/fonts/,
  Extension=.ttf,
  UprightFont=*,
  BoldFont=*,
  BoldFeatures={FakeBold=1.7},
  ItalicFont=*,
  ItalicFeatures={FakeSlant=0.12},
  BoldItalicFont=*,
  BoldItalicFeatures={FakeBold=1.7,FakeSlant=0.12}
]{LXGWNeoXiHei}

\setsansfont[
  Path=assets/fonts/,
  Extension=.ttf,
  UprightFont=*,
  BoldFont=*,
  BoldFeatures={FakeBold=1.7},
  ItalicFont=*,
  ItalicFeatures={FakeSlant=0.12},
  BoldItalicFont=*,
  BoldItalicFeatures={FakeBold=1.7,FakeSlant=0.12}
]{LXGWNeoXiHei}

\setmonofont[
  Path=assets/fonts/,
  Extension=.ttf,
  UprightFont=*-Regular,
  BoldFont=*-Bold,

  BoldItalicFont=*-BoldItalic,
  Scale=0.90
]{LiberationMono}

% spaces; the selected primary families cover both Latin and CJK.
